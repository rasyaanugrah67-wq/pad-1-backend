<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class CheckRole
{
    public function handle(
        Request $request,
        Closure $next,
        ...$roles
    ): Response {

        $akun = $request->user();

        if (!$akun) {
            return response()->json([
                'success' => false,
                'message' => 'Unauthenticated.',
            ], 401);
        }

        if (!$akun->isActive()) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak aktif.',
            ], 403);
        }

        $roleName = strtolower(
            $akun->role?->nama_role ?? ''
        );

        $allowedRoles = array_map(
            'strtolower',
            $roles
        );

        if (!in_array($roleName, $allowedRoles, true)) {
            return response()->json([
                'success' => false,
                'message' => 'Anda tidak memiliki akses ke resource ini.',
            ], 403);
        }

        return $next($request);
    }
}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\AnggotaPendaftaran;
use App\Models\Lomba;
use App\Models\Notifikasi;
use App\Models\Pendaftaran;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Validator;

class PendaftaranController extends Controller
{
    /**
     * PB-04: Mengambil riwayat pendaftaran milik warga yang sedang login (Individu & Beregu).
     */
    public function index(Request $request): JsonResponse
    {
        $akun = $request->user();
        $idWarga = $akun->id_warga;

        if (!$idWarga) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak terhubung dengan data warga.',
            ], 403);
        }

        $pendaftaran = Pendaftaran::with([
            'lomba' => function ($query) {
                $query->with([
                    'kategori',
                    'jadwal' => function ($q) {
                        $q->orderBy('tanggal')->orderBy('waktu_mulai');
                    },
                ]);
            },
            'anggota',
        ])
            ->where('id_warga', $idWarga)
            ->orderByDesc('tanggal_daftar')
            ->orderByDesc('id_pendaftaran')
            ->get();

        return response()->json([
            'success' => true,
            'message' => 'Daftar riwayat pendaftaran lomba berhasil diambil.',
            'data' => $pendaftaran,
        ]);
    }

    /**
     * PB-04: Mendaftar lomba (Mendukung Individu dan Regu/Tim berdasarkan data lomba di DB).
     */
    public function store(Request $request): JsonResponse
    {
        $akun = $request->user();
        $idWarga = $akun->id_warga;

        if (!$idWarga) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak terhubung dengan data warga.',
            ], 403);
        }

        // 1. Validasi awal: id_lomba wajib integer
        $initialValidator = Validator::make($request->all(), [
            'id_lomba' => ['required', 'integer'],
        ], [
            'id_lomba.required' => 'ID lomba wajib diisi.',
            'id_lomba.integer' => 'ID lomba harus berupa angka.',
        ]);

        if ($initialValidator->fails()) {
            return response()->json([
                'success' => false,
                'message' => $initialValidator->errors()->first(),
                'errors' => $initialValidator->errors(),
            ], 422);
        }

        // 2. Ambil data lomba dari database dan periksa status
        $lomba = Lomba::with('kategori')->find($request->id_lomba);

        if (!$lomba) {
            return response()->json([
                'success' => false,
                'message' => 'Lomba tidak ditemukan.',
            ], 422);
        }

        if ($lomba->status !== 'Dibuka') {
            return response()->json([
                'success' => false,
                'message' => "Pendaftaran tidak dapat dilakukan karena status lomba saat ini '{$lomba->status}'.",
            ], 422);
        }

        // 3. Tentukan jenis pendaftaran berdasarkan data lomba di database
        $isBeregu = $lomba->isBeregu()
            || (strtolower($lomba->jenis_pendaftaran ?? '') === 'beregu')
            || in_array($lomba->nama_lomba, ['Tarik Tambang', 'Futsal Sarung'], true);

        // 4. Validasi berdasarkan jenis pendaftaran lomba
        if ($isBeregu) {
            // Lomba beregu WAJIB menyertakan nama_tim dan daftar anggota
            $teamValidator = Validator::make($request->all(), [
                'nama_tim' => ['required', 'string', 'min:2', 'max:100'],
                'anggota' => ['required', 'array', 'min:1'],
                'anggota.*.nama' => ['required', 'string', 'max:100'],
                'anggota.*.no_hp' => [
                    'required',
                    'string',
                    'max:20',
                    'regex:/^(\+62|62|0)[0-9\-\s]{7,18}$/',
                ],
            ], [
                'nama_tim.required' => 'Nama tim wajib diisi untuk lomba beregu.',
                'nama_tim.string' => 'Nama tim harus berupa teks.',
                'nama_tim.max' => 'Nama tim maksimal 100 karakter.',
                'anggota.required' => 'Daftar anggota tim wajib diisi untuk lomba beregu.',
                'anggota.array' => 'Daftar anggota harus berupa array.',
                'anggota.min' => 'Daftar anggota tim minimal memiliki 1 anggota.',
                'anggota.*.nama.required' => 'Nama setiap anggota tim wajib diisi.',
                'anggota.*.nama.string' => 'Nama anggota harus berupa teks.',
                'anggota.*.nama.max' => 'Nama anggota maksimal 100 karakter.',
                'anggota.*.no_hp.required' => 'Nomor HP setiap anggota tim wajib diisi.',
                'anggota.*.no_hp.regex' => 'Format nomor HP anggota tidak valid (contoh: 081234567890).',
            ]);

            if ($teamValidator->fails()) {
                return response()->json([
                    'success' => false,
                    'message' => $teamValidator->errors()->first(),
                    'errors' => $teamValidator->errors(),
                ], 422);
            }
        } else {
            // Lomba individu CUKUP id_lomba. Tolak jika mengirim field nama_tim atau anggota
            $hasNamaTim = $request->filled('nama_tim');
            $hasAnggota = $request->has('anggota') && !empty($request->input('anggota'));

            if ($hasNamaTim || $hasAnggota) {
                return response()->json([
                    'success' => false,
                    'message' => 'Lomba ini adalah lomba individu, tidak menerima data nama tim atau anggota.',
                ], 422);
            }
        }

        // 5. Cegah perwakilan mendaftar dua kali untuk lomba yang sama
        $sudahTerdaftar = Pendaftaran::where('id_warga', $idWarga)
            ->where('id_lomba', $lomba->id_lomba)
            ->exists();

        if ($sudahTerdaftar) {
            return response()->json([
                'success' => false,
                'message' => 'Anda sudah terdaftar pada perlombaan ini.',
            ], 409);
        }

        // 6. Simpan pendaftaran, anggota tim, dan notifikasi dalam satu DB Transaction
        $namaTim = $isBeregu ? trim($request->input('nama_tim')) : null;
        $anggotaList = $isBeregu ? $request->input('anggota', []) : [];

        $pendaftaran = DB::transaction(function () use ($idWarga, $lomba, $isBeregu, $namaTim, $anggotaList) {
            $p = Pendaftaran::create([
                'id_warga' => $idWarga,
                'id_lomba' => $lomba->id_lomba,
                'nama_tim' => $namaTim,
                'tanggal_daftar' => now(),
                'status' => 'Menunggu',
                'catatan' => null,
            ]);

            if ($isBeregu) {
                foreach ($anggotaList as $item) {
                    AnggotaPendaftaran::create([
                        'id_pendaftaran' => $p->id_pendaftaran,
                        'nama' => trim($item['nama']),
                        'no_hp' => trim($item['no_hp']),
                    ]);
                }
            }

            // Notifikasi pendaftaran
            $pesanNotif = $isBeregu
                ? "Pendaftaran tim '{$namaTim}' untuk lomba '{$lomba->nama_lomba}' telah diajukan dan sedang menunggu verifikasi panitia."
                : "Pendaftaran Anda untuk lomba '{$lomba->nama_lomba}' telah diajukan dan sedang menunggu verifikasi panitia.";

            Notifikasi::create([
                'id_warga' => $idWarga,
                'judul' => 'Pendaftaran Lomba Berhasil Diajukan',
                'pesan' => $pesanNotif,
                'tipe' => 'Lomba',
                'status_baca' => false,
            ]);

            return $p;
        });

        // 7. Load relasi lomba, kategori, jadwal, dan anggota
        $pendaftaran->load([
            'lomba' => function ($query) {
                $query->with([
                    'kategori',
                    'jadwal' => function ($q) {
                        $q->orderBy('tanggal')->orderBy('waktu_mulai');
                    },
                ]);
            },
            'anggota',
        ]);

        return response()->json([
            'success' => true,
            'message' => $isBeregu
                ? "Pendaftaran tim '{$namaTim}' berhasil diajukan."
                : 'Pendaftaran lomba berhasil diajukan.',
            'data' => $pendaftaran,
        ], 201);
    }
}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Iuran;
use App\Models\Notifikasi;
use App\Models\Pembayaran;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\Rule;

class IuranController extends Controller
{
    /**
     * PB-05: Mengambil daftar iuran milik warga yang sedang login.
     */
    public function index(Request $request): JsonResponse
    {
        $akun = $request->user();
        $idWarga = $akun->id_warga;

        if (!$idWarga) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak terhubung dengan data warga.',
            ], 403);
        }

        $validated = $request->validate([
            'status' => [
                'nullable',
                Rule::in(['Semua', 'Belum Bayar', 'Sudah Bayar']),
            ],
        ]);

        $status = $validated['status'] ?? 'Semua';

        $query = Iuran::with([
            'periode',
            'pembayaran' => function ($q) {
                $q->orderByDesc('created_at');
            },
        ])
            ->where('id_warga', $idWarga);

        if ($status === 'Belum Bayar') {
            $query->where('status', 'Belum Bayar');
        } elseif ($status === 'Sudah Bayar') {
            $query->where('status', 'Sudah Bayar');
        }

        $iuran = $query
            ->orderBy('jatuh_tempo')
            ->orderByDesc('id_iuran')
            ->get();

        return response()->json([
            'success' => true,
            'message' => 'Daftar tagihan iuran berhasil diambil.',
            'data' => $iuran,
        ]);
    }

    /**
     * PB-05: Mengambil detail satu iuran milik warga yang sedang login.
     */
    public function show(Request $request, int $id_iuran): JsonResponse
    {
        $akun = $request->user();
        $idWarga = $akun->id_warga;

        if (!$idWarga) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak terhubung dengan data warga.',
            ], 403);
        }

        $iuran = Iuran::with([
            'periode',
            'pembayaran' => function ($q) {
                $q->with(['verifikator' => function ($v) {
                    $v->select('id_akun', 'username');
                }])->orderByDesc('created_at');
            },
        ])
            ->where('id_iuran', $id_iuran)
            ->where('id_warga', $idWarga)
            ->first();

        if (!$iuran) {
            return response()->json([
                'success' => false,
                'message' => 'Data iuran tidak ditemukan.',
            ], 404);
        }

        return response()->json([
            'success' => true,
            'message' => 'Detail iuran berhasil diambil.',
            'data' => $iuran,
        ]);
    }

    /**
     * PB-05: Mengajukan pembayaran untuk suatu tagihan iuran.
     */
    public function bayar(Request $request, int $id_iuran): JsonResponse
    {
        $akun = $request->user();
        $idWarga = $akun->id_warga;

        if (!$idWarga) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak terhubung dengan data warga.',
            ], 403);
        }

        // Cari iuran dan pastikan milik warga yang login
        $iuran = Iuran::where('id_iuran', $id_iuran)
            ->where('id_warga', $idWarga)
            ->first();

        if (!$iuran) {
            return response()->json([
                'success' => false,
                'message' => 'Data iuran tidak ditemukan.',
            ], 404);
        }

        // Cek jika status iuran sudah lunas
        if ($iuran->status === 'Sudah Bayar') {
            return response()->json([
                'success' => false,
                'message' => 'Iuran ini sudah lunas.',
            ], 422);
        }

        // Cek apakah ada pengajuan pembayaran sebelumnya yang masih menunggu verifikasi
        $menungguVerifikasi = Pembayaran::where('id_iuran', $iuran->id_iuran)
            ->where('status_verifikasi', 'Menunggu')
            ->exists();

        if ($menungguVerifikasi) {
            return response()->json([
                'success' => false,
                'message' => 'Pengajuan pembayaran sebelumnya masih menunggu verifikasi panitia.',
            ], 409);
        }

        $request->validate([
            'metode' => ['required', 'string', 'max:50'],
            'nominal' => ['required', 'numeric', 'min:1'],
            'bukti_bayar' => ['required', 'file', 'image', 'mimes:jpeg,png,jpg,webp', 'max:2048'],
        ]);

        // Pastikan nominal pembayaran sama dengan tagihan iuran
        if (round((float) $request->nominal, 2) !== round((float) $iuran->nominal, 2)) {
            return response()->json([
                'success' => false,
                'message' => 'Nominal pembayaran harus sesuai dengan tagihan iuran (Rp ' . number_format($iuran->nominal, 0, ',', '.') . ').',
            ], 422);
        }

        // Simpan file bukti bayar ke disk 'public'
        $path = $request->file('bukti_bayar')->store('bukti_bayar', 'public');

        $pembayaran = DB::transaction(function () use ($iuran, $request, $path, $idWarga) {
            $p = Pembayaran::create([
                'id_iuran' => $iuran->id_iuran,
                'tanggal_bayar' => now(),
                'nominal' => $iuran->nominal,
                'metode' => $request->metode,
                'bukti_bayar' => $path,
                'status_verifikasi' => 'Menunggu',
                'diverifikasi_oleh' => null,
                'tanggal_verifikasi' => null,
            ]);

            Notifikasi::create([
                'id_warga' => $idWarga,
                'judul' => 'Pengajuan Pembayaran Iuran',
                'pesan' => 'Pembayaran iuran sebesar Rp ' . number_format($iuran->nominal, 0, ',', '.') . ' berhasil diajukan dan sedang menunggu verifikasi panitia.',
                'tipe' => 'Iuran',
                'status_baca' => false,
            ]);

            return $p;
        });

        return response()->json([
            'success' => true,
            'message' => 'Pembayaran iuran berhasil diajukan dan sedang menunggu verifikasi panitia.',
            'data' => $pembayaran,
        ], 201);
    }
}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Notifikasi;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class NotifikasiController extends Controller
{
    /**
     * PB-06: Mengambil seluruh notifikasi milik warga yang login.
     */
    public function index(Request $request): JsonResponse
    {
        $akun = $request->user();
        $idWarga = $akun->id_warga;

        if (!$idWarga) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak terhubung dengan data warga.',
            ], 403);
        }

        $query = Notifikasi::where('id_warga', $idWarga)
            ->orderByDesc('created_at')
            ->orderByDesc('id_notifikasi');

        if ($request->has('status_baca')) {
            $statusBaca = filter_var($request->query('status_baca'), FILTER_VALIDATE_BOOLEAN, FILTER_NULL_ON_FAILURE);
            if ($statusBaca !== null) {
                $query->where('status_baca', $statusBaca);
            }
        }

        $notifikasi = $query->get();

        return response()->json([
            'success' => true,
            'message' => 'Daftar notifikasi berhasil diambil.',
            'data' => $notifikasi,
        ]);
    }

    /**
     * PB-06: Mengambil jumlah notifikasi yang belum dibaca (unread count).
     */
    public function unreadCount(Request $request): JsonResponse
    {
        $akun = $request->user();
        $idWarga = $akun->id_warga;

        if (!$idWarga) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak terhubung dengan data warga.',
            ], 403);
        }

        $unreadCount = Notifikasi::where('id_warga', $idWarga)
            ->where('status_baca', false)
            ->count();

        return response()->json([
            'success' => true,
            'message' => 'Jumlah notifikasi belum dibaca berhasil diambil.',
            'data' => [
                'unread_count' => $unreadCount,
            ],
        ]);
    }

    /**
     * PB-06: Menandai satu notifikasi telah dibaca (idempotent).
     */
    public function markAsRead(Request $request, int $id_notifikasi): JsonResponse
    {
        $akun = $request->user();
        $idWarga = $akun->id_warga;

        if (!$idWarga) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak terhubung dengan data warga.',
            ], 403);
        }

        $notifikasi = Notifikasi::where('id_notifikasi', $id_notifikasi)
            ->where('id_warga', $idWarga)
            ->first();

        if (!$notifikasi) {
            return response()->json([
                'success' => false,
                'message' => 'Notifikasi tidak ditemukan.',
            ], 404);
        }

        if (!$notifikasi->status_baca) {
            $notifikasi->update([
                'status_baca' => true,
            ]);
        }

        return response()->json([
            'success' => true,
            'message' => 'Notifikasi berhasil ditandai sudah dibaca.',
            'data' => $notifikasi,
        ]);
    }

    /**
     * PB-06: Menandai semua notifikasi milik warga sebagai sudah dibaca.
     */
    public function markAllAsRead(Request $request): JsonResponse
    {
        $akun = $request->user();
        $idWarga = $akun->id_warga;

        if (!$idWarga) {
            return response()->json([
                'success' => false,
                'message' => 'Akun tidak terhubung dengan data warga.',
            ], 403);
        }

        $affected = Notifikasi::where('id_warga', $idWarga)
            ->where('status_baca', false)
            ->update([
                'status_baca' => true,
            ]);
